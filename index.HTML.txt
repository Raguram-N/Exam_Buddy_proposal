let metricQueriesCount = 12;
let metricTopicsRevised = 8;

// Dynamic Speech Framework Settings
const SpeechRecognitionEngine = window.SpeechRecognition || window.webkitSpeechRecognition;
const nativeAudioSynthesizer = window.speechSynthesis;
let isVoiceSystemActive = false;
let browserSpeechCaptureInstance = null;

if (SpeechRecognitionEngine) {
browserSpeechCaptureInstance = new SpeechRecognitionEngine();
browserSpeechCaptureInstance.continuous = false;
browserSpeechCaptureInstance.lang = 'en-IN';
browserSpeechCaptureInstance.interimResults = false;

browserSpeechCaptureInstance.onstart = () => {
isVoiceSystemActive = true;
document.getElementById("vOuterRingNode").classList.add("active-pulse");

document.getElementById("vCenterDotNode").innerHTML = "🛑";

};

browserSpeechCaptureInstance.onresult = (event) => {
const spokenTranscriptionResult = event.results[0][0].transcript;
fireStudyPipeline(spokenTranscriptionResult);
};

browserSpeechCaptureInstance.onerror = (err) => {
console.error("Audio capture error logged:", err.error);
terminateVoiceSystemMode();
};

browserSpeechCaptureInstance.onend = () => {
terminateVoiceSystemMode();
};
}

function toggleVoiceSystemProcessingLink() {
if (!browserSpeechCaptureInstance) {
alert("Native Web Speech API tracking is not exposed inside this browser interface. Please use keyboard entry.");
return;
}
if (isVoiceSystemActive) {
browserSpeechCaptureInstance.stop();
} else {
nativeAudioSynthesizer.cancel();
browserSpeechCaptureInstance.start();
}
}

function terminateVoiceSystemMode() {
isVoiceSystemActive = false;
document.getElementById("vOuterRingNode").classList.remove("active-pulse");
document.getElementById("vCenterDotNode").classList.remove("active-recording");
document.getElementById("vCenterDotNode").innerHTML = "🎙️";
document.getElementById("voiceStatusLabelText").innerHTML = "Voice Engine Standby.<br>Click to engage.";
}

}
